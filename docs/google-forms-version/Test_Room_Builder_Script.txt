/**
 * TEST ROOM REPORT - ONE-CLICK BUILDER  (Google Apps Script)
 *
 * This ONE script does everything:
 *   - creates the Google Form "Test Room Report" with all rooms K-0 to K-50
 *   - creates the linked Google Sheet for answers
 *   - makes a Google Doc with one QR code per room (print and stick)
 *   - makes a Desk QR page for principal / coordinator (open or download latest reports)
 *   - sends an instant alert for any "No", any non-attempting student, or counts that don't add up
 *   - 20 minutes after the test starts (8:10), sends a list of rooms that did NOT report
 *
 * HOW TO USE (on a computer / laptop, signed in to the account that should own everything):
 *   1) Go to script.google.com  ->  New project
 *   2) Delete the sample code, paste ALL of this
 *   3) Check the SETTINGS below
 *   4) At the top choose function "buildEverything" and press Run
 *      (allow permissions once: Advanced -> Go to project -> Allow)
 *   5) Open "Execution log" - it shows the links to your Form, Sheet and QR Docs
 * Running buildEverything again is safe: it will not make a second form.
 */

// ===== SETTINGS =====
const ALERT_EMAILS     = 'zamankips@gmail.com';   // comma-separate for more people
const VIEWER_EMAILS    = '';                      // coordinator(s) who may open/download reports, comma-separated
const TIMEZONE         = 'Asia/Karachi';          // India: 'Asia/Kolkata'
const TELEGRAM_TOKEN   = '';                      // optional; leave '' to use email only
const TELEGRAM_CHAT_ID = '';                      // optional
const TEST_START       = '07:50';                 // test start time (24-hour, hh:mm)
const BUFFER_MINUTES   = 20;                      // "no report" alert this many minutes after start
const TEST_DAYS        = [1, 3, 5];               // Mon=1, Tue=2, Wed=3, Thu=4, Fri=5, Sat=6, Sun=7
const SKIP_DATES       = [];                      // holidays, e.g. ['2026-11-16'] (yyyy-mm-dd)

// All rooms K-0 to K-50. Rooms that hold a test have their section here.
const SECTIONS = {
  'K-4':'F-1A','K-5':'F-1B','K-6':'F-2','K-7':'F-3','K-3':'F-5','K-2':'F-6','K-1':'F-7','K-11':'F-8',
  'K-9':'F-9','K-14':'F-10','K-15':'F-11','K-16':'F-12','K-13':'F-13','K-10':'F-14',
  'K-8':'F-15','K-44':'F-16',
  'K-39':'S1A','K-40':'S1B','K-43':'S2','K-50':'S3','K-49':'S4','K-27':'S6','K-23':'S7','K-25':'S8',
  'K-28':'S9','K-32':'S10','K-45':'S11','K-46':'S12','K-22':'S13','K-29':'S14','K-31':'S15','K-48':'S16'
};
const FIRST_ROOM = 0, LAST_ROOM = 50;

const ROOMS = [];        // every room: gets a QR and is in the dropdown
const TEST_ROOMS = [];   // only these are checked at 8:10 for "not reported"
for (let n = FIRST_ROOM; n <= LAST_ROOM; n++) {
  const k = 'K-' + n;
  const label = SECTIONS[k] ? k + ' | ' + SECTIONS[k] : k;
  ROOMS.push(label);
  if (SECTIONS[k]) TEST_ROOMS.push(label);
}

// Form question titles
const F = { room:'Room', name:'Your name', role:'Reporting as',
            total:'Total students', present:'Present students', absent:'Absent students',
            late:'Late arrivals', nonatt:'Non-attempting students',
            names:'Names of non-attempting students', remarks:'Other remarks' };
const CHECKS = ['Whiteboard written?','Seating arranged?','A/B paper distributed properly?',
                'Margin line intimation to students?','Underlining intimation to students?',
                'Highlighting intimation to students?'];
const FI_CHECK = 'Teacher intimation about lagging?';   // only shown to Floor Incharge
const ALL_CHECKS = CHECKS.concat([FI_CHECK]);

const PROPS = PropertiesService.getScriptProperties();

// ===== RUN THIS ONCE =====
function buildEverything() {
  let form;
  if (PROPS.getProperty('FORM_ID')) {
    form = FormApp.openById(PROPS.getProperty('FORM_ID'));
    Logger.log('Form already exists, not making a new one.');
  } else {
    form = createForm_();
  }

  let ss;
  if (PROPS.getProperty('SHEET_ID')) {
    ss = SpreadsheetApp.openById(PROPS.getProperty('SHEET_ID'));
  } else {
    ss = SpreadsheetApp.create('Test Room Reports (answers)');
    form.setDestination(FormApp.DestinationType.SPREADSHEET, ss.getId());
    PROPS.setProperty('SHEET_ID', ss.getId());
  }
  if (VIEWER_EMAILS) ss.addViewers(VIEWER_EMAILS.split(',').map(s => s.trim()).filter(String));

  const docUrl = PROPS.getProperty('QR_DOC_URL') || makeQrDoc_(form);
  const deskUrl = PROPS.getProperty('DESK_DOC_URL') || makeDeskQrDoc_(ss);
  installTriggers_(ss);

  Logger.log('DONE');
  Logger.log('Form (to edit):    ' + form.getEditUrl());
  Logger.log('Form (to fill):    ' + form.getPublishedUrl());
  Logger.log('Answers Sheet:     ' + ss.getUrl());
  Logger.log('Room QR codes:     ' + docUrl);
  Logger.log('Desk QR (office):  ' + deskUrl);
}

// ===== RUN THIS AFTER CHANGING THE ROOM LIST =====
// Updates the Room dropdown in the same form and makes a fresh QR Doc (old one goes to Bin).
function updateRooms() {
  const form = FormApp.openById(PROPS.getProperty('FORM_ID'));
  form.getItems(FormApp.ItemType.LIST)[0].asListItem().setChoiceValues(ROOMS);

  const oldUrl = PROPS.getProperty('QR_DOC_URL') || '';
  const oldId = (oldUrl.match(/id=([\w-]+)/) || oldUrl.match(/\/d\/([\w-]+)/) || [])[1];
  if (oldId) DriveApp.getFileById(oldId).setTrashed(true);
  PROPS.deleteProperty('QR_DOC_URL');

  Logger.log('Rooms in form: ' + ROOMS.length + '   Checked at 8:10: ' + TEST_ROOMS.length);
  Logger.log('NEW QR codes to print: ' + makeQrDoc_(form));
}

// Page 1: room, name, role.  Floor Incharge -> page 2 (lagging question) -> page 3.
// Invigilator -> straight to page 3 (checks and counts).
function createForm_() {
  const form = FormApp.create('Test Room Report');
  form.setDescription('Fill this during the test period (7:50 - 8:40). Choose Yes or No honestly.')
      .setAllowResponseEdits(false)
      .setCollectEmail(false)
      .setConfirmationMessage('Thank you. Your report has been sent.');

  form.addListItem().setTitle(F.room).setChoiceValues(ROOMS).setRequired(true);
  form.addTextItem().setTitle(F.name).setRequired(true);
  const role = form.addMultipleChoiceItem().setTitle(F.role).setRequired(true);

  const fiPage = form.addPageBreakItem().setTitle('Floor Incharge check');
  form.addMultipleChoiceItem().setTitle(FI_CHECK).setChoiceValues(['Yes', 'No']).setRequired(true);

  const mainPage = form.addPageBreakItem().setTitle('Room checks');
  CHECKS.forEach(c => form.addMultipleChoiceItem().setTitle(c)
      .setChoiceValues(['Yes', 'No']).setRequired(true));

  const wholeNumber = () => FormApp.createTextValidation()
      .setHelpText('Type a number: 0, 1, 2 ...')
      .requireTextMatchesPattern('^[0-9]+$').build();
  [F.total, F.present, F.absent, F.late, F.nonatt].forEach(t => form.addTextItem().setTitle(t)
      .setValidation(wholeNumber()).setRequired(true));

  form.addParagraphTextItem().setTitle(F.names)
      .setHelpText('Write names only if non-attempting is more than 0').setRequired(false);
  form.addParagraphTextItem().setTitle(F.remarks).setRequired(false);

  role.setChoices([role.createChoice('Invigilator', mainPage),
                   role.createChoice('Floor Incharge', fiPage)]);

  PROPS.setProperty('FORM_ID', form.getId());
  return form;
}

function qrBlob_(text) {
  return UrlFetchApp.fetch('https://quickchart.io/qr?size=600&margin=2&text=' +
                           encodeURIComponent(text)).getBlob();
}

// One page per room: big room name + QR code that opens the form with the room already chosen
function makeQrDoc_(form) {
  const roomItem = form.getItems(FormApp.ItemType.LIST)[0].asListItem();
  const doc = DocumentApp.create('Test Room QR Codes (print)');
  const body = doc.getBody();
  body.clear();

  ROOMS.forEach((room, i) => {
    const url = form.createResponse()
        .withItemResponse(roomItem.createResponse(room))
        .toPrefilledUrl();

    const title = body.appendParagraph('ROOM ' + room);
    title.setHeading(DocumentApp.ParagraphHeading.TITLE)
         .setAlignment(DocumentApp.HorizontalAlignment.CENTER);
    const imgPara = body.appendParagraph('');
    imgPara.setAlignment(DocumentApp.HorizontalAlignment.CENTER);
    imgPara.appendInlineImage(qrBlob_(url)).setWidth(380).setHeight(380);
    body.appendParagraph('Test Room Report\nScan at 7:50 and submit')
        .setAlignment(DocumentApp.HorizontalAlignment.CENTER)
        .editAsText().setFontSize(20);
    if (i < ROOMS.length - 1) body.appendPageBreak();
  });

  doc.saveAndClose();
  PROPS.setProperty('QR_DOC_URL', doc.getUrl());
  return doc.getUrl();
}

// Desk page: one QR opens the live Sheet, one downloads the latest Excel file
function makeDeskQrDoc_(ss) {
  const doc = DocumentApp.create('Desk QR - Test Reports (print)');
  const body = doc.getBody();
  body.clear();
  const items = [
    ['OPEN LIVE REPORTS', ss.getUrl()],
    ['DOWNLOAD LATEST (Excel)', 'https://docs.google.com/spreadsheets/d/' + ss.getId() + '/export?format=xlsx']
  ];
  body.appendParagraph('Test Invigilation Reports')
      .setHeading(DocumentApp.ParagraphHeading.TITLE)
      .setAlignment(DocumentApp.HorizontalAlignment.CENTER);
  items.forEach(([label, url]) => {
    body.appendParagraph(label).setHeading(DocumentApp.ParagraphHeading.HEADING1)
        .setAlignment(DocumentApp.HorizontalAlignment.CENTER);
    const p = body.appendParagraph('');
    p.setAlignment(DocumentApp.HorizontalAlignment.CENTER);
    p.appendInlineImage(qrBlob_(url)).setWidth(230).setHeight(230);
  });
  body.appendParagraph('Only works for Google accounts the reports are shared with.')
      .setAlignment(DocumentApp.HorizontalAlignment.CENTER);
  doc.saveAndClose();
  PROPS.setProperty('DESK_DOC_URL', doc.getUrl());
  return doc.getUrl();
}

function installTriggers_(ss) {
  ScriptApp.getProjectTriggers().forEach(t => ScriptApp.deleteTrigger(t));
  ScriptApp.newTrigger('onFormSubmit').forSpreadsheet(ss).onFormSubmit().create();
  ScriptApp.newTrigger('scheduleTodayCheck').timeBased()
      .everyDays(1).atHour(5).inTimezone(TIMEZONE).create();
  scheduleTodayCheck();
}

// Runs early every morning: sets the "no report" check for exactly TEST_START + BUFFER_MINUTES today
function scheduleTodayCheck() {
  ScriptApp.getProjectTriggers()
      .filter(t => t.getHandlerFunction() === 'checkMissingReports')
      .forEach(t => ScriptApp.deleteTrigger(t));
  const now = new Date();
  const day = Utilities.formatDate(now, TIMEZONE, 'yyyy-MM-dd');
  const offset = Utilities.formatDate(now, TIMEZONE, 'XXX');            // e.g. +05:00
  const start = new Date(day + 'T' + TEST_START + ':00' + offset);
  const when = new Date(start.getTime() + BUFFER_MINUTES * 60000);
  if (when > now) ScriptApp.newTrigger('checkMissingReports').timeBased().at(when).create();
}

// ===== RUNS ON EVERY FORM SUBMIT =====
function onFormSubmit(e) {
  const v = e.namedValues;
  const get = k => (v[k] && v[k][0] ? String(v[k][0]).trim() : '');
  const num = k => Number(get(k)) || 0;
  const failed = ALL_CHECKS.filter(c => get(c).toLowerCase() === 'no');
  const nonatt = num(F.nonatt);
  const countsWrong = num(F.present) + num(F.absent) !== num(F.total);
  if (failed.length === 0 && nonatt === 0 && !countsWrong) return;   // all fine: stay silent

  let msg = 'ALERT  ' + get(F.room) + '\n' +
            'By: ' + get(F.name) + ' (' + get(F.role) + ')\n';
  if (failed.length) msg += 'Problems: ' + failed.join(', ') + '\n';
  if (nonatt > 0)    msg += 'Non-attempting students: ' + nonatt +
                            (get(F.names) ? ' (' + get(F.names) + ')' : '') + '\n';
  msg += 'Total: ' + num(F.total) + '   Present: ' + num(F.present) +
         '   Absent: ' + num(F.absent) + '   Late: ' + num(F.late) + '\n';
  if (countsWrong)   msg += 'Note: Present + Absent does not equal Total\n';
  if (get(F.remarks)) msg += 'Remarks: ' + get(F.remarks) + '\n';
  notify('ALERT ' + get(F.room), msg);
}

// ===== RUNS AT TEST_START + BUFFER_MINUTES (8:10) : which rooms have not reported? =====
function checkMissingReports() {
  const now = new Date();
  const today = Utilities.formatDate(now, TIMEZONE, 'yyyy-MM-dd');
  const weekday = Number(Utilities.formatDate(now, TIMEZONE, 'u'));   // Mon=1 ... Sun=7
  if (TEST_DAYS.indexOf(weekday) < 0 || SKIP_DATES.indexOf(today) >= 0) return;

  const ss = SpreadsheetApp.openById(PROPS.getProperty('SHEET_ID'));
  const sh = ss.getSheets().filter(s => s.getFormUrl())[0] || ss.getSheets()[0];
  const data = sh.getDataRange().getValues();
  const roomCol = data[0].indexOf(F.room);
  const reported = {};
  for (let i = 1; i < data.length; i++) {
    const ts = data[i][0];
    if (ts instanceof Date && Utilities.formatDate(ts, TIMEZONE, 'yyyy-MM-dd') === today)
      reported[String(data[i][roomCol]).trim()] = true;
  }
  const missing = TEST_ROOMS.filter(r => !reported[r]);
  if (missing.length === 0) notify('All rooms reported', 'All ' + TEST_ROOMS.length + ' test rooms have reported today.');
  else notify('NO REPORT from ' + missing.length + ' rooms',
              'Not yet reported today (' + today + '):\n' + missing.join('\n'));
}

// ===== SENDING =====
function notify(subject, body) {
  MailApp.sendEmail(ALERT_EMAILS, subject, body);
  if (TELEGRAM_TOKEN && TELEGRAM_CHAT_ID) {
    UrlFetchApp.fetch('https://api.telegram.org/bot' + TELEGRAM_TOKEN + '/sendMessage',
      { method: 'post', payload: { chat_id: TELEGRAM_CHAT_ID, text: subject + '\n' + body } });
  }
}

// ===== OPTIONAL: run this to check that alerts reach your phone =====
function testAlert() {
  notify('TEST alert', 'If you see this, alerts are working.');
}
